import json import difflib # خواندن پایگاه داده سایت‌ها از فایل sites.json def load_database(filepath="sites.json"): try: with open(filepath, "r", encoding="utf-8") as f: return json.load(f) except Exception as e: print(f"خطا در خواندن فایل: {e}") return [] # موتور جستجوی پیشرفته محلی def smart_search(query, sites): query = query.strip().lower() if not query: return [] results = [] query_tokens = query.split() for site in sites: title = site.get("title", "") url = site.get("url", "") score = 0 title_lower = title.lower() url_lower = url.lower() # تطابق دقیق کل عبارت if query in title_lower: score += 100 if query in url_lower: score += 50 # تطابق کلمات به صورت مجزا (جستجوی فازی و بخشی) for token in query_tokens: if token in title_lower: score += 30 # استفاده از تطابق فازی برای مقاومت در برابر غلط املایی match_ratio = difflib.SequenceMatcher(None, token, title_lower).find_longest_match(0, len(token), 0, len(title_lower)) if match_ratio.size > 2: score += match_ratio.size * 5 if score > 0: results.append({"site": site, "score": score}) # مرتب‌سازی بر اساس بیشترین امتیاز (مرتبط‌ترین‌ها اول بیان) results.sort(key=lambda x: x["score"], reverse=True) # خروجی دادن فقط لیست سایت‌ها بدون امتیاز return [r["site"] for r in results] # تست موتور جستجو if __name__ == "__main__": database = load_database("sites.json") while True: user_query = input("\nجستجو در MEEPNO (برای خروج q): ") if user_query.lower() == 'q': break search_results = smart_search(user_query, database) print(f"\nنتایج برای: '{user_query}' ({len(search_results)} مورد پیدا شد):") print("-" * 40) if search_results: for i, res in enumerate(search_results[:10], 1): print(f"{i}. {res['title']}") print(f" 🔗 {res['url']}") else: print("چیزی پیدا نشد داداش!")
